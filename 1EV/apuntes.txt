Apuntes Docker
- Hacer docker interactivo --> docker run --name web -ti --hostname web ubuntu:latest 
- Hacer docker con volumen --> docker run --name [name] --hostname [name] -ti -p [xxxx:xx] ubuntu:latest -v /home/alumno/web:/var/www/html --name web3 --hostname web3 web:v2

- Instalar apache en el docker --> apt install apache2 php libapache2-mod-php
- Hacer commit de la imagen
  - docker commit -m "mensaje" [nombre_contenedor] [nombre:version]

#Borrar todos los contenedores
- docker rm -f $(docker ps -a -q)

#Instalar mediante el dockerfile
- docker run -ti --name [nombre] --hostname [nombre] -p [puerto] -v [volumen] [imagen]
- docker run -ti --name web10 --hostname web -p 8100:80 -v /home/alumno/2DAW/DWES/:/var/www/html web:v1

# Configuracion para aplicarla en todos los directorios (dice que las configuraciones hagan que si no encuentra el archivo

RUN echo "<Directory /var/www/html>" >> /etc/apache2/sites-available/000-default.conf
RUN echo "Options Indexes" >> /etc/apache2/sites-available/000-default.conf
RUN echo "</Directory>" >> /etc/apache2/sites-available/000-default.conf
